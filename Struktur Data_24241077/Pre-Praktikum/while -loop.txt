# while-loop
'''
    syntax :
    while kondisi(TRUE): # selama bernilai TRUE
     aksi
'''
angka = 0
print(f"angka awal -> (angka)")

while angka < 5:
    angka += 1
    print(f"angka sekarang -> (angka)")
print("keluar dari looping")
